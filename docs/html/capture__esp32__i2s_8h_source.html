23// Global (single-instance) I2S DMA capture state - kept at namespace scope,
24// like the rest of this library's singletons (pin_reader_ptr, buffer_ptr),
25// so the interrupt handler below can be a plain function instead of a class
26// member (member functions placed in per-TU COMDAT sections have caused
27// "literal placed after use" link errors for IRAM_ATTR ISRs on some
28// esp32-arduino core versions).
29static volatile bool i2s_capture_dma_done = false;
30static intr_handle_t i2s_capture_intr_handle = nullptr;

32static void IRAM_ATTR i2s_capture_isr(void *arg) {
33 if (I2S0.int_raw.in_suc_eof) {
34 i2s_capture_dma_done = true;
35 esp_intr_disable(i2s_capture_intr_handle);
36 I2S0.conf.rx_start = 0;
37 }
38 I2S0.int_clr.val = I2S0.int_raw.val;
39}

187 void allocDma(uint32_t sampleCount) {
188 freeDma();
189 // Each 32 bit DMA word carries 2 samples (see dma_elem_t layout below),
190 // so we need sampleCount/2 32-bit words, rounded up to whole descriptors.
191 size_t total_bytes = ((sampleCount + 1) / 2) * 4;
192 dma_desc_count = (total_bytes + desc_size - 1) / desc_size;
193 if (dma_desc_count == 0) dma_desc_count = 1;
194
195 dma_buf = (uint8_t **) malloc(sizeof(uint8_t *) * dma_desc_count);
196 dma_desc = (lldesc_t *) malloc(sizeof(lldesc_t) * dma_desc_count);
197
198 for (size_t i = 0; i < dma_desc_count; i++) {
199 dma_buf[i] = (uint8_t *) malloc(desc_size);
200 lldesc_t *pd = &dma_desc[i];
201 pd->length = desc_size;
202 pd->size = desc_size;
203 pd->owner = 1;
204 pd->sosf = 1;
205 pd->buf = dma_buf[i];
206 pd->offset = 0;
207 pd->empty = 0;
208 bool last = (i + 1 == dma_desc_count);
209 pd->eof = last ? 1 : 0;
210 pd->qe.stqe_next = last ? nullptr : &dma_desc[i + 1];
211 }
212 }

214 void freeDma() {
215 if (dma_buf != nullptr) {
216 for (size_t i = 0; i < dma_desc_count; i++) free(dma_buf[i]);
217 free(dma_buf);
218 dma_buf = nullptr;
219 }
220 if (dma_desc != nullptr) {
221 free(dma_desc);
222 dma_desc = nullptr;
223 }
224 dma_desc_count = 0;
225 }

234 void start() {
235 setupOnce();
236 n_samples = logicAnalyzer().readCount();
237 allocDma(n_samples);
238 // clear stale heap contents so a missing/too-slow sample clock (see class
239 // comment) is reported as a flat 0 capture instead of leaking old heap data
240 for (size_t i = 0; i < dma_desc_count; i++) memset(dma_buf[i], 0, desc_size);
241
242 uint32_t freq = logicAnalyzer().captureFrequency();
243 enableSampleClock(freq);
244
245 i2s_capture_dma_done = false;
246 i2sConfReset();
247 I2S0.rx_eof_num = (n_samples + 1) / 2; // in 32 bit words (2 samples/word)
248 I2S0.in_link.addr = (uint32_t) &dma_desc[0];
249 I2S0.in_link.start = 1;
250 I2S0.int_clr.val = I2S0.int_raw.val;
251 I2S0.int_ena.val = 0;
252 I2S0.int_ena.in_suc_eof = 1;
253
254 run_time_us = 0;
255 start_time = micros();
256 esp_intr_enable(i2s_capture_intr_handle);
257 I2S0.conf.rx_start = 1;
258 }

260 void waitForResult() {
261 log("waitForResult()");
262 // the datasheet-documented rx_bck_div_num=1 corner case + slow clocks can
263 // take a while; bound the wait so a bad frequency can't hang forever.
264 unsigned long deadline = millis() + 5000;
265 while (!i2s_capture_dma_done && millis() < deadline) delay(1);
266 run_time_us = micros() - start_time;
267 ledcDetach(clk_out_pin);
268
269 RingBuffer &buf = logicAnalyzer().buffer();
270 buf.clear();
271 size_t written = 0;
272 for (size_t i = 0; i < dma_desc_count && written < n_samples; i++) {
273 uint8_t *b = dma_buf[i];
274 // 32 bit DMA words hold two samples as 00 s2 00 s1 (see dma_elem_t in
275 // the reference implementation) - byte 1 is the first sample, byte 3
276 // (== byte 1 of the next word) is the second.
277 for (size_t off = 0; off + 3 < desc_size && written < n_samples; off += 4) {
278 buf.write((PinBitArray) b[off + 1]);
279 written++;
280 if (written < n_samples) {
281 buf.write((PinBitArray) b[off + 3]);
282 written++;
283 }
284 }
285 }
286 log("waitForResult() -> %u records", (unsigned) buf.available());
287 }
