64void write(PinBitArray bits) {
65 // The SUMP/OLS protocol expects each 4 byte sample in the host's native
66 // (little endian) byte order - NOT network (big endian) order, so we must
67 // not apply htonl() here. Also Arduino's Print::write(unsigned long)
68 // truncates to a single byte, so we write the full 4 bytes explicitly.
69 uint32_t tmp = (uint32_t) bits;
70 stream_ptr->write((const uint8_t*)&tmp, sizeof(tmp));
71}

73// writes a buffer of uint32_t values
74void write(uint32_t *buff, size_t n_samples) {
75 int written = 0;
76 int open = n_samples * sizeof(uint32_t);
77 while(open > 0){
78 size_t result = stream_ptr->write((const char*)buff + written, open);
79 written += result;
80 open -= result;
81 }
82}

84// writes a buffer of PinBitArray
85void write(PinBitArray *buff, size_t n_samples) {
86 // convert to uint32_t - samples must stay in native (little endian) byte
87 // order to match what the SUMP/OLS protocol (and PulseView) expects.
88 uint32_t tmp[DUMP_RECORD_SIZE];
89 int idx = 0;
90 for (int j=0;j<n_samples;j++){
91 tmp[idx++] = (uint32_t) buff[j];
92 if (idx==DUMP_RECORD_SIZE){
93 write(tmp, idx);
94 idx = 0;
95 }
96 }
97 if (idx>0){
98 write(tmp, idx);
99 }
100}

455 void capture(bool is_max_speed) {
456 log("capture is_max_speed: %s", is_max_speed ? "true":"false");
457 // waiting for trigger
458 if (la_state.trigger_mask) {
459 log("waiting for trigger");
460 while ((la_state.trigger_values ^ captureSample()) & la_state.trigger_mask)
461 ;
462 }
463 la_state.setStatus(TRIGGERED);
464 log("triggered");
465
466 // remove unnecessary entries from buffer based on delayCount & readCount
467 long keep = la_state.read_count - la_state.delay_count;
468 if (keep > 0 && buffer_ptr->available()>keep) {
469 log("keeping last %ld entries",keep);
470 buffer_ptr->clear(buffer_ptr->available() - keep);
471 } else if (keep < 0) {
472 log("ignoring first %ld entries",abs(keep));
473 buffer_ptr->clear(buffer_ptr->available() + abs(keep));
474 } else if (keep==0l){
475 log("starting with clean buffer");
476 buffer_ptr->clear();
477 }
478

480 if (is_max_speed){
481 if (la_state.is_continuous_capture){
482 captureAllContinousMaxSpeed();
483 } else {
484 captureAllMaxSpeed();
485 dumpData();
486 log("capture-done: %lu",buffer_ptr->available());

488 }
489 } else {

491 captureAllContinous();
492 } else {
493 captureAll();

876 void processCommand(int cmd){
877
878 switch (cmd) {
882 case SUMP_RESET:
883 // debounce reset
884 if (millis()>sump_reset_igorne_timeout){
885 log("=>SUMP_RESET");
886 setStatus(STOPPED);
887 clear();
888 sump_reset_igorne_timeout = millis()+ 500;
889 raiseEvent(RESET);
890 }
891 break;
892
896 case SUMP_ID:
897 log("=>SUMP_ID");
898 stream().write(device_id, 4);
899 stream().flush();
900 break;
901
902 /*
903 * We return a description of our capabilities.
904 * Check the function's comments below.
905 */
906 case SUMP_GET_METADATA:
907 log("=>SUMP_GET_METADATA");
908 sendMetadata();
909 break;
910
911 /*
912 * Captures the data
913 */
914 case SUMP_ARM:
915 log("=>SUMP_ARM");
916 // clear current data
917 clear();
918 setStatus(ARMED);
919 if (is_capture_on_arm){
920 capture();
921 }
922 break;
923
924 /*
925 * the trigger mask byte has a '1' for each enabled trigger so
926 * we can just use it directly as our trigger mask.
927 */
928 case SUMP_TRIGGER_MASK:
929 log("=>SUMP_TRIGGER_MASK");
930 setTriggerMask(commandExtPinBitArray());
931 break;
932
933 /*
934 * trigger_values can be used directly as the value of each bit
935 * defines whether we're looking for it to be high or low.
936 */
937 case SUMP_TRIGGER_VALUES:
938 log("=>SUMP_TRIGGER_VALUES");
939 setTriggerValues(commandExtPinBitArray());
940 break;
941
942
943 /* read the rest of the command bytes but ignore them. */
944 case SUMP_TRIGGER_CONFIG:
945 log("=>SUMP_TRIGGER_CONFIG");
946 commandExt();
947 break;
948
949 /*
950 * the shifting needs to be done on the 32bit unsigned long variable
951 * so that << 16 doesn't end up as zero.
952 */
953 case SUMP_SET_DIVIDER: {
954 log("=>SUMP_SET_DIVIDER");
955 Sump4ByteComandArg cmd = commandExt();
956 uint32_t divider = cmd.get32();
957 log("-divider: %lu\n", divider);
958 setupDelay(divider);
959 }
960 break;
961
962 /*
963 * this just sets up how many samples there should be before
964 * and after the trigger fires. The readCount is total samples
965 * to return and delayCount number of samples after the trigger.
966 * this sets the buffer splits like 0/100, 25/75, 50/50
967 * for example if readCount == delayCount then we should
968 * return all samples starting from the trigger point.
969 * if delayCount < readCount we return (readCount - delayCount) of
970 * samples from before the trigger fired.
971 */
972 case SUMP_SET_READ_DELAY_COUNT: {
973 Sump4ByteComandArg cmd = commandExt();
974 log("=>SUMP_SET_READ_DELAY_COUNT %02X %02X",cmd.get16(0),cmd.get16(1));
975 la_state.read_count = (cmd.get16(0)+1) * 4;
976 la_state.delay_count = (cmd.get16(1)+1) * 4;
977 log("--> read_count: %d", la_state.read_count);
978 log("--> delay_count: %d", la_state.delay_count);
979 raiseEvent(READ_DLEAY_COUNT);
980 }
981 break;
982
983 /* read the rest of the command bytes and check if RLE is enabled. */
984 case SUMP_SET_FLAGS: {
985 log("=>SUMP_SET_FLAGS");
986 Sump4ByteComandArg cmd = commandExt();
987 la_state.is_continuous_capture = ((cmd.getPtr()[1] & 0B1000000) != 0);
988 log("--> is_continuous_capture: %d\n", la_state.is_continuous_capture);
989 raiseEvent(FLAGS);
990
991 }
992 break;
993
994 /* ignore any unrecognized bytes. */
995 default:
996 log("=>UNHANDLED command: %d", cmd);
997 break;
998
999 };
