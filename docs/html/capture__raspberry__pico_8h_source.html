156 void arm() {
157 log("arm()");
158
159 log("- Init trigger");
160
161 // Grant high bus priority to the DMA, so it can shove the processors out
162 // of the way. This should only be needed if you are pushing things up to
163 // >16bits/clk here, i.e. if you need to saturate the bus completely.
164 bus_ctrl_hw->priority = BUSCTRL_BUS_PRIORITY_DMA_W_BITS | BUSCTRL_BUS_PRIORITY_DMA_R_BITS;
165
166 // Load a program to capture n pins. This is just a single `in pins, n`
167 // instruction with a wrap.
168 uint16_t capture_prog_instr = pio_encode_in(pio_pins, pin_count);
169 struct pio_program capture_prog = {
170 .instructions = &capture_prog_instr,
171 .length = 1,
172 .origin = -1
173 };
174
175 uint offset = pio_add_program(pio, &capture_prog);
176
177 // Route the requested GPIOs to the PIO block and set them as inputs.
178 // Without this the state machine never sees the external pin levels
179 // (it stays connected to SIO), so the capture just reads zeros.
180 for (uint i = 0; i < pin_count; i++) {
181 pio_gpio_init(pio, pin_base + i);
182 }
183 pio_sm_set_consecutive_pindirs(pio, sm, pin_base, pin_count, false);
184
185 // Configure state machine to loop over this `in` instruction forever,
186 // with autopush enabled.
187 pio_sm_config c = pio_get_default_sm_config();
188 sm_config_set_in_pins(&c, pin_base);
189 sm_config_set_wrap(&c, offset, offset);
190 sm_config_set_clkdiv(&c, divider_value);
191 // Note that we may push at a < 32 bit threshold if pin_count does not
192 // divide 32. We are using shift-to-right, so the sample data ends up
193 // left-justified in the FIFO in this case, with some zeroes at the LSBs.
194 sm_config_set_in_shift(&c, true, true, 32);
195 sm_config_set_fifo_join(&c, PIO_FIFO_JOIN_RX);
196 pio_sm_init(pio, sm, offset, &c);
197
199 log("- Arming trigger");
200 pio_sm_set_enabled(pio, sm, false);
201 // Need to clear _input shift counter_, as well as FIFO, because there may be
202 // partial ISR contents left over from a previous run. sm_restart does this.
203 pio_sm_clear_fifos(pio, sm);
204 pio_sm_restart(pio, sm);
205
206 dma_channel_config dma_config = dma_channel_get_default_config(dma_chan);
207 channel_config_set_read_increment(&dma_config, false);
208 channel_config_set_write_increment(&dma_config, true);
209 channel_config_set_transfer_data_size(&dma_config, DMA_SIZE_32);
210 channel_config_set_dreq(&dma_config, pio_get_dreq(pio, sm, false));
211
212 n_transfers = n_samples /4 * sizeof(PinBitArray);
213 dma_channel_configure(dma_chan, &dma_config,
214 logicAnalyzer().buffer().data_ptr(), // Destination pointer
215 &pio->rxf[sm], // Source pointer
216 n_transfers, // Number of transfers
217 true // Start immediately
218 );
219
222
223 run_time_us = 0;
224 start_time = micros();
225 pio_sm_set_enabled(pio, sm, true);
226 }
